Relational logical schema

SOCIETA (id_societa, nome, provincia*, regione_provenienza*, nazione_provenienza*, email_referente*, telefono_referente*)

CATEGORIA (id_categoria, formato_gioco, tipo_categoria)

TORNEO (id_torneo, nome, data_inizio, data_fine, luogo)

GIOCATORE (id_persona, data_nascita, nome, cognome)

GENITORE (id_persona, con_figlio, grado_parentela, nome, cognome)

HOTEL (id_hotel, nome, indirizzo, distanza, telefono*, email*)

SQUADRA (id_squadra, nome, SOCIETA_id_societa, CATEGORIA_id_categoria)

ISCRIZIONE (id_iscrizione, data_iscrizione, stato, note*, SQUADRA_id_squadra, TORNEO_id_torneo, HOTEL_id_hotel*)

PARTECIPAZIONE (id_partecipazione, ISCRIZIONE_id_iscrizione, GIOCATORE_id_persona, TARIFFA_id_tariffa*)

TARIFFA (id_tariffa, costo, descrizione, formula_roulette*, TORNEO_id_torneo)

STANZA (id_stanza, numero, posti_letto, HOTEL_id_hotel)

RISERVARE (ISCRIZIONE_id_iscrizione, STANZA_id_stanza)

METTERE_A_DISPOSIZIONE (TORNEO_id_torneo, STANZA_id_stanza)

OCCUPARE_GIOCATORE (GIOCATORE_id_persona, STANZA_id_stanza, data_inizio, data_fine)

OCCUPARE_GENITORE (GENITORE_id_persona, STANZA_id_stanza, data_inizio, data_fine)
