Relational logical schema

HOTEL (id_hotel, nome, indirizzo, distanza, telefono*, email*)

SOCIETA (id_societa, nome, regione_provenienza*, nazione_provenienza*, email_referente*, telefono_referente*)

GIOCATORE (id_persona, data_nascita, nome, cognome)

STANZA (id_stanza, numero, posti_letto)

GENITORE (id_persona, con_figlio, grado_parentela, nome, cognome)

TORNEO (id_torneo, nome, data_inizio, data_fine, luogo)

CATEGORIA (id_categoria, formato_gioco, tipo_categoria)

SQUADRA (id_squadra, nome, SOCIETA_id_societa, CATEGORIA_id_categoria)

TARIFFA (id_tariffa, costo, descrizione, TORNEO_id_torneo)

ISCRIZIONE (id_iscrizione, data_iscrizione, stato, note*, TORNEO_id_torneo, SQUADRA_id_squadra)

AVERE (SQUADRA_id_squadra, GIOCATORE_id_persona)

FORNIRE (HOTEL_id_hotel, STANZA_id_stanza, TORNEO_id_torneo)

SCEGLIERE (GIOCATORE_id_persona, TARIFFA_id_tariffa)

RISERVARE (STANZA_id_stanza, ISCRIZIONE_id_iscrizione)

COMPORRE (GIOCATORE_id_persona, ISCRIZIONE_id_iscrizione)

OCCUPARE_GIOCATORE (GIOCATORE_id_persona, STANZA_id_stanza)

OCCUPARE_GENITORE (STANZA_id_stanza, GENITORE_id_persona)

ISCRIZIONE (SQUADRA_id_squadra) REFERENCES SQUADRA (id_squadra)

AVERE (SQUADRA_id_squadra) REFERENCES SQUADRA (id_squadra)

AVERE (GIOCATORE_id_persona) REFERENCES GIOCATORE (id_persona)

FORNIRE (HOTEL_id_hotel) REFERENCES HOTEL (id_hotel)

FORNIRE (STANZA_id_stanza) REFERENCES STANZA (id_stanza)

RISERVARE (STANZA_id_stanza) REFERENCES STANZA (id_stanza)

RISERVARE (ISCRIZIONE_id_iscrizione) REFERENCES ISCRIZIONE (id_iscrizione)

COMPORRE (GIOCATORE_id_persona) REFERENCES GIOCATORE (id_persona)

COMPORRE (ISCRIZIONE_id_iscrizione) REFERENCES ISCRIZIONE (id_iscrizione)

OCCUPARE_GIOCATORE (STANZA_id_stanza) REFERENCES STANZA (id_stanza)
