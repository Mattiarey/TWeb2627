/* Entities */
entity HOTEL {
    id_hotel (id),
    nome,
    indirizzo,
    distanza,
    telefono (optional),
    email (optional)
}
entity SQUADRA {
    id_squadra (id),
    nome
}
entity SOCIETA {
    id_societa (id),
    nome,
    regione_provenienza (optional),
    nazione_provenienza (optional),
    email_referente (optional),
    telefono_referente (optional)
}
entity GIOCATORE {
    data_nascita
}
entity STANZA {
    numero,
    posti_letto,
    id_stanza (id)
}
entity PERSONA {
    id_persona (id),
    nome,
    cognome
}
entity GENITORE {
    con_figlio,
    grado_parentela
}
entity TORNEO {
    id_torneo (id),
    nome,
    data_inizio,
    data_fine,
    luogo
}
entity TARIFFA {
    costo,
    descrizione,
    id_tariffa (id)
}
entity CATEGORIA {
    id_categoria (id),
    formato_gioco,
    tipo_categoria
}
entity ISCRIZIONE {
    id_iscrizione (id),
    data_iscrizione,
    stato,
    note (optional)
}

/* Relationships */
relationship CONTENERE (
    SOCIETA: one..many,
    SQUADRA: one..one
)
relationship AVERE (
    SQUADRA: one..many,
    GIOCATORE: zero..many
)
relationship FORNIRE (
    HOTEL: zero..many,
    STANZA: zero..many,
    TORNEO: zero..many
)
relationship OCCUPARE (
    STANZA: zero..many,
    PERSONA: zero..many
) {
    data_inizio,
    data_fine
}
relationship PROPORRE (
    TORNEO: one..many,
    TARIFFA: one..one
)
relationship SCEGLIERE (
    GIOCATORE: one..many,
    TARIFFA: zero..many
)
relationship GIOCARE_IN (
    CATEGORIA: one..many,
    SQUADRA: one..one
)
relationship ISCRIVERE (
    ISCRIZIONE: one..one,
    TORNEO: one..many
)
relationship SOTTOPORRE (
    ISCRIZIONE: one..one,
    SQUADRA: zero..many
)
relationship RISERVARE (
    ISCRIZIONE: one..many,
    STANZA: zero..many
)
relationship COMPORRE (
    GIOCATORE: zero..many,
    ISCRIZIONE: zero..many
)

/* Generalizations */
PERSONA <= {
    GIOCATORE,
    GENITORE
} (total, exclusive)
